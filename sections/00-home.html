Clase y Constructor

        public string $nombre,

        return $this->nombre;

echo $u->getNombre(); // Ana
Herencia e Interfaces
<?php
interface Comparable {
    public function compareTo(self $other): int;
}

class Producto implements Comparable {
    public function __construct(
        public string $nombre,
        public float $precio,
    ) {}

    public function compareTo(self $other): int {
        return $this->precio <=> $other->precio;
    }
}

$a = new Producto("A", 10);
$b = new Producto("B", 20);
echo $a->compareTo($b); // -1
PHP 8.1+ Enums
<?php
enum Estado: string {
    case Activo   = "activo";
    case Inactivo = "inactivo";
    case Pendiente = "pendiente";

    public function label(): string {
        return match($this) {
            self::Activo   => "Activo",
            self::Inactivo => "Inactivo",
            self::Pendiente => "Pendiente",
        };
    }
}

$e = Estado::Activo;
echo $e->value;    // "activo"
echo $e->label();  // "Activo"

Conexion PDO
<?php
try {
    $pdo = new PDO(
        "mysql:host=localhost;dbname=tienda;charset=utf8mb4",
        "root",
        "",
        [
            PDO::ATTR_ERRMODE => PDO::ERRMODE_EXCEPTION,
            PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
        ]
    );
    echo "Conexion exitosa!";
} catch (PDOException $e) {
    die("Error: " . $e->getMessage());
}
Prepared Statements
<?php
// INSERT seguro con prepared statements
$stmt = $pdo->prepare(
    "INSERT INTO usuarios (nombre, email, password)
     VALUES (:nombre, :email, :pass)"
);

$stmt->execute([
    ':nombre' => $nombre,
    ':email'  => $email,
    ':pass'   => password_hash($pass, PASSWORD_DEFAULT),
]);

echo "Usuario creado con ID: " . $pdo->lastInsertId();
Transacciones
<?php
try {
    $pdo->beginTransaction();

    $pdo->exec("UPDATE cuentas SET saldo = saldo - 100 WHERE id = 1");
    $pdo->exec("UPDATE cuentas SET saldo = saldo + 100 WHERE id = 2");

    $pdo->commit();
    echo "Transferencia exitosa!";
} catch (Exception $e) {
    $pdo->rollBack();
    echo "Error: " . $e->getMessage();
}
Seguridad Web

Protege tu aplicacion

XSS, CSRF, SQL Injection, hash de contrasenas y las mejores practicas de seguridad.

Ver seccion completa
XSS - Cross Site Scripting
<?php
// PELIGRO: Output sin sanitizar
echo $_GET['nombre']; // <script>alert('XSS')</script>

// SEGURO: Escapar output
echo htmlspecialchars($_GET['nombre'], ENT_QUOTES, 'UTF-8');

// SEGURO: Usar Content Security Policy
header("Content-Security-Policy: default-src 'self'");
Contrasenas Seguras
<?php
// NUNCA guardar contrasenas en texto plano
$hash = password_hash("mi_contrasena", PASSWORD_DEFAULT);
// $2y$10$N9qo8uLOickgx2ZMRZoMy...

// Verificar contrasena
if (password_verify("mi_contrasena", $hash)) {
    echo "Contrasena correcta!";
}

// Verificar si necesita rehash (actualizar algorithm)
if (password_needs_rehash($hash, PASSWORD_DEFAULT)) {
    $nuevoHash = password_hash("mi_contrasena", PASSWORD_DEFAULT);
}
CSRF Token
<?php
// Generar token
session_start();
if (empty($_SESSION['csrf_token'])) {
    $_SESSION['csrf_token'] = bin2hex(random_bytes(32));
}

// En el formulario HTML:
// <input type="hidden" name="csrf_token"
//        value="<?= $_SESSION['csrf_token'] ?>">

// Verificar en el POST
if ($_POST['csrf_token'] !== $_SESSION['csrf_token']) {
    die("Token CSRF invalido!");
}

Named Arguments
<php
function create_user(
    string $name,
    string $email,
    int $age = 0,
    bool $active = true
): array {
    return compact("name", "email", "age", "active");
}

// Sin named args (confuso)
create_user("Ana", "ana@mail.com", 25, false);

// Con named args (claro y flexible)
create_user(
    name: "Ana",
    email: "ana@mail.com",
    active: false,
);
Fibers (PHP 8.1)
<php
