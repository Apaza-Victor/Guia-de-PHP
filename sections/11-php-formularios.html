validar.php
<?php
function validarFormulario(array $datos): array {
    $errores = [];

    // Nombre: requerido, 2–50 chars, solo letras y espacios
    $nombre = trim($datos["nombre"] ?? "");
    if (empty($nombre)) {
        $errores["nombre"] = "El nombre es obligatorio";
    } elseif (strlen($nombre) < 2 || strlen($nombre) > 50) {
        $errores["nombre"] = "Entre 2 y 50 caracteres";
    } elseif (!preg_match('/^[\p{L}\s]+$/u', $nombre)) {
        $errores["nombre"] = "Solo letras y espacios";
    }

    // Email: requerido y formato válido
    $email = trim($datos["email"] ?? "");
    if (empty($email)) {
        $errores["email"] = "El email es obligatorio";
    } elseif (!filter_var($email, FILTER_VALIDATE_EMAIL)) {
        $errores["email"] = "Formato de email inválido";
    }

    // Edad: número entero 1–120
    $edad = filter_var($datos["edad"] ?? "", FILTER_VALIDATE_INT,
        ["options" => ["min_range" => 1, "max_range" => 120]]
    );
    if ($edad === false) {
        $errores["edad"] = "Edad debe ser un número entre 1 y 120";
    }

    // Rol: solo valores permitidos
    $rolesValidos = ["admin", "editor", "lector"];
    $rol = $datos["rol"] ?? "";
    if (!in_array($rol, $rolesValidos, true)) {
        $errores["rol"] = "Rol no válido";
    }

    return $errores;
}

if ($_SERVER["REQUEST_METHOD"] === "POST") {
    $errores = validarFormulario($_POST);
    if (empty($errores)) {
        echo "✅ Formulario válido, procesando...";
    } else {
        foreach ($errores as $campo => $error) {
            echo "❌ $campo: $error<br>";
        }
    }
}
?>
▶ Salida si hay errores
❌ nombre: Solo letras y espacios
❌ email: Formato de email inválido
✅ Formulario válido, procesando...
filter_var.php
<?php
// VALIDAR — devuelve el valor si es válido, false si no
filter_var("ana@mail.com", FILTER_VALIDATE_EMAIL);   // "ana@mail.com"
filter_var("no-es-email",   FILTER_VALIDATE_EMAIL);   // false
filter_var("https://php.net", FILTER_VALIDATE_URL);     // "https://php.net"
filter_var("192.168.1.1",    FILTER_VALIDATE_IP);      // "192.168.1.1"
filter_var("42",             FILTER_VALIDATE_INT);     // 42
filter_var("3.14",           FILTER_VALIDATE_FLOAT);   // 3.14
filter_var("true",           FILTER_VALIDATE_BOOLEAN); // true

// Con opciones
filter_var(150, FILTER_VALIDATE_INT, [
    "options" => ["min_range" => 1, "max_range" => 100]
]);  // false — 150 > 100

// SANITIZAR — limpia el valor (siempre devuelve string)
filter_var("<h1>Hola!</h1>",   FILTER_SANITIZE_SPECIAL_CHARS);
// "&lt;h1&gt;Hola!&lt;/h1&gt;" — escapa HTML

filter_var(" ana@mail.com  ",  FILTER_SANITIZE_EMAIL);
// "ana@mail.com" — elimina caracteres no permitidos en emails

filter_var("hola mundo <script>", FILTER_SANITIZE_SPECIAL_CHARS);
// "hola mundo &lt;script&gt;"

// filter_input — lee directamente de superglobal (más seguro)
$email = filter_input(INPUT_POST, "email", FILTER_VALIDATE_EMAIL);
$pagina = filter_input(INPUT_GET,  "page",  FILTER_VALIDATE_INT);

// Sanitizar manualmente (siempre hacer esto antes de mostrar texto de usuario)
$textoSeguro = htmlspecialchars($_POST["comentario"] ?? "", ENT_QUOTES, "UTF-8");
?>
▶ Referencia rápida
FILTER_VALIDATE_EMAIL → valor o false
FILTER_SANITIZE_SPECIAL_CHARS → escapa < > &
filter_input() lee POST/GET y valida en un paso
upload.php
<?php
// HTML del formulario (enctype es obligatorio para archivos)
// <form method="POST" enctype="multipart/form-data">
//   <input type="file" name="imagen" accept="image/*">
// </form>

if (isset($_FILES["imagen"])) {
    $archivo = $_FILES["imagen"];

    // Información disponible en $_FILES
    echo $archivo["name"];      // nombre original del archivo
    echo $archivo["type"];      // MIME type reportado (NO fiable)
    echo $archivo["size"];      // tamaño en bytes
    echo $archivo["tmp_name"];  // ruta temporal en servidor
    echo $archivo["error"];     // 0 = sin error

    // Validaciones de seguridad
    $maxSize   = 2 * 1024 * 1024;  // 2 MB
    $tiposOk  = ["image/jpeg", "image/png", "image/webp"];
    $extOk    = ["jpg", "jpeg", "png", "webp"];

    // 1. Sin errores
    if ($archivo["error"] !== UPLOAD_ERR_OK) {
        throw new RuntimeException("Error al subir");
    }
    // 2. Tamaño
    if ($archivo["size"] > $maxSize) {
        throw new RuntimeException("Archivo demasiado grande");
    }
    // 3. Tipo MIME real (no el reportado por el cliente)
    $finfo    = new finfo(FILEINFO_MIME_TYPE);
    $mimeReal = $finfo->file($archivo["tmp_name"]);
    if (!in_array($mimeReal, $tiposOk, true)) {
        throw new RuntimeException("Tipo de archivo no permitido");
    }
    // 4. Extensión segura
    $ext = strtolower(pathinfo($archivo["name"], PATHINFO_EXTENSION));
    if (!in_array($ext, $extOk, true)) {
        throw new RuntimeException("Extensión no permitida");
    }

    // Nombre aleatorio para evitar colisiones y ataques
    $nuevoNombre = bin2hex(random_bytes(16)) . "." . $ext;
    $destino     = __DIR__ . "/uploads/" . $nuevoNombre;

    if (move_uploaded_file($archivo["tmp_name"], $destino)) {
        echo "✅ Imagen guardada: $nuevoNombre";
    }
}
?>
▶ Salida
✅ Imagen guardada: a3f8c1d2e4b6...webp
💡 Guarda los archivos fuera del DocumentRoot o en una carpeta sin ejecución PHP. Nunca uses el nombre original del archivo del usuario.
