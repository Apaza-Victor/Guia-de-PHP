tipos_error.php
<?php
// E_NOTICE — variable no definida
echo $variableNoDefinida;
// Notice: Undefined variable $variableNoDefinida

// E_WARNING — archivo no existe
include "archivo_inexistente.php";
// Warning: include(): Failed opening 'archivo_inexistente.php'

// E_DEPRECATED — función obsoleta
mysql_connect();  // Deprecated en PHP 5, eliminada en PHP 7

// E_ERROR — clase no existe (detiene el script)
$obj = new ClaseQueNoExiste();
// Fatal error: Class "ClaseQueNoExiste" not found

// Generar error manualmente
trigger_error("Algo salió mal", E_USER_WARNING);
trigger_error("Error grave",       E_USER_ERROR);
?>
config_errores.php
<?php
// ── DESARROLLO ──────────────────────────────────
error_reporting(E_ALL);             // todos los errores
ini_set('display_errors', '1');      // mostrar en pantalla
ini_set('display_startup_errors', '1');

// ── PRODUCCIÓN ──────────────────────────────────
error_reporting(E_ALL & ~E_NOTICE & ~E_DEPRECATED);
ini_set('display_errors', '0');      // NUNCA mostrar en pantalla
ini_set('log_errors',     '1');      // guardar en archivo
ini_set('error_log',      '/var/log/php_errors.log');

// Registrar error manualmente en el log
error_log("Error crítico en el pago del usuario #123");

// Controlar qué errores mostrar con máscaras de bits
error_reporting(E_ERROR | E_WARNING);  // solo fatales y advertencias
error_reporting(E_ALL & ~E_NOTICE);  // todos excepto notices

// También en php.ini:
// error_reporting = E_ALL
// display_errors = Off
// log_errors = On
// error_log = /var/log/php_errors.log
?>

error_handler.php
<?php
// set_error_handler — capturar errores de PHP con función propia
set_error_handler(function(int $nivel, string $msg, string $archivo, int $linea): bool {
    $tipos = [E_WARNING => "⚠️ WARNING", E_NOTICE => "📌 NOTICE", E_USER_ERROR => "🔴 ERROR"];
    $tipo  = $tipos[$nivel] ?? "❓ UNKNOWN";
    $log   = sprintf("[%s] %s — %s en %s línea %d\n",
        date('Y-m-d H:i:s'), $tipo, $msg, basename($archivo), $linea
    );
    error_log($log);
    return true;  // true = no ejecutar el handler por defecto de PHP
});

// set_exception_handler — capturar excepciones no capturadas
set_exception_handler(function(Throwable $e): void {
    error_log("Excepción no capturada: " . $e->getMessage());
    http_response_code(500);
    echo json_encode(['error' => 'Error interno del servidor']);
});

// register_shutdown_function — ejecutar al terminar el script
register_shutdown_function(function(): void {
    $error = error_get_last();
    if ($error && in_array($error['type'], [E_ERROR, E_PARSE])) {
        error_log("Fatal: " . $error['message']);
    }
});
?>
▶ Entrada en el log
[2024-01-15 10:30:00] ⚠️ WARNING — file not found en index.php línea 12

Try / Catch / Finally y Excepciones
Manejo estructurado · excepciones personalizadas · jerarquía · rethrow · finally.
Diagrama de flujo: try / catch / finally
try_catch.php
<?php
function dividir(float $a, float $b): float {
    if ($b == 0) {
        throw new InvalidArgumentException("No se puede dividir por cero", 400);
    }
    return $a / $b;
}

try {
    echo dividir(10, 2);   // 5
    echo dividir(10, 0);   // lanza excepción
    echo "Esta línea NO se ejecuta";

} catch (InvalidArgumentException $e) {
    // Información de la excepción
    echo "Mensaje: "  . $e->getMessage();   // No se puede dividir por cero
    echo "Código: "   . $e->getCode();      // 400
    echo "Archivo: "  . basename($e->getFile());
    echo "Línea: "    . $e->getLine();

} catch (TypeError | ValueError $e) {
    // Captura múltiple con | (PHP 8)
    echo "Error de tipo/valor: " . $e->getMessage();

} catch (Throwable $e) {
    // Throwable captura CUALQUIER error o excepción
    echo "Error inesperado";

} finally {
    // Se ejecuta SIEMPRE — ideal para liberar recursos
    echo "Script finalizado (finally)";
}
?>
▶ Salida
5
Mensaje: No se puede dividir por cero | Código: 400
Script finalizado (finally)
excepciones_propias.php
<?php
// Jerarquía de excepciones propias
class AppException        extends RuntimeException {}
class ValidacionException extends AppException {
    public function __construct(
        string $msg,
        private string $campo = ""
    ) {
        parent::__construct($msg, 422);
    }
    public function getCampo(): string { return $this->campo; }
}
class AuthException       extends AppException {}
class NotFoundException   extends AppException {}

function registrarUsuario(array $datos): void {
    if (empty($datos['email'])) {
        throw new ValidacionException("El email es obligatorio", "email");
    }
    if (strlen($datos['password'] ?? "") < 8) {
        throw new ValidacionException("Contraseña muy corta (min 8)", "password");
    }
}

try {
    registrarUsuario(['email' => '', 'password' => '123']);

} catch (ValidacionException $e) {
    echo "Campo: {$e->getCampo()} — {$e->getMessage()}<br>";
} catch (AuthException $e) {
    echo "Sin permisos: {$e->getMessage()}<br>";
} catch (AppException $e) {
    echo "Error de app: {$e->getMessage()}<br>";
}
?>
▶ Salida
Campo: email — El email es obligatorio
rethrow_jerarquia.php
<?php
// Jerarquía de Throwable en PHP
// Throwable
// ├── Error (errores internos de PHP)
// │   ├── TypeError, ValueError, ArithmeticError...
// └── Exception (lanzadas con throw)
// │   ├── RuntimeException, LogicException, InvalidArgumentException...
// │   └── (tus excepciones personalizadas)

// Rethrow — capturar, registrar y relanzar
function conectarBD(): PDO {
    try {
        return new PDO("mysql:host=localhost;dbname=test", "root", "");
    } catch (PDOException $e) {
        error_log("DB Error: " . $e->getMessage());
        // Relanzar como excepción más genérica
        throw new RuntimeException("No se pudo conectar a la BD", 500, $e);
    }
}

try {
    $pdo = conectarBD();

} catch (RuntimeException $e) {
    echo $e->getMessage();       // "No se pudo conectar a la BD"

    // Excepción original (PDOException) disponible en getPrevious()
    $original = $e->getPrevious();
    echo "Causa original: " . $original?->getMessage();
}

// Excepciones nativas más usadas
// InvalidArgumentException — argumento inválido
// RuntimeException        — error en tiempo de ejecución
// LogicException          — error en la lógica del programa
// OverflowException       — desbordamiento
// OutOfRangeException     — índice fuera de rango
// BadMethodCallException  — método no existe o mal llamado
?>

No se pudo conectar a la BD
Causa original: SQLSTATE[HY000] [2002] Connection refused
