SQL Fundamental
Los comandos SQL que todo desarrollador PHP debe conocer.
esquema.sql
-- Crear base de datos
CREATE DATABASE tienda CHARACTER SET utf8mb4 COLLATE utf8mb4_unicode_ci;
USE tienda;

-- Crear tabla
CREATE TABLE productos (
    id          INT AUTO_INCREMENT PRIMARY KEY,
    nombre      VARCHAR(150) NOT NULL,
    precio      DECIMAL(10,2) NOT NULL,
    stock       INT DEFAULT 0,
    categoria   VARCHAR(80),
    creado_en   TIMESTAMP DEFAULT CURRENT_TIMESTAMP
);

-- Insertar datos
INSERT INTO productos (nombre, precio, stock, categoria)
VALUES ('Laptop Pro', 1299.99, 15, 'Electrónica'),
       ('Teclado Mec.', 89.00, 42, 'Accesorios');

-- Consultas
SELECT * FROM productos;
SELECT nombre, precio FROM productos WHERE precio > 100;
SELECT * FROM productos ORDER BY precio DESC LIMIT 10;

-- Actualizar
UPDATE productos SET precio = 1199.99 WHERE id = 1;

-- Eliminar
DELETE FROM productos WHERE id = 2;

PDO — Conexión segura a MySQL
PDO (PHP Data Objects) es el método recomendado para conectar PHP con bases de datos. Previene SQL Injection con sentencias preparadas.
🔐 Nunca uses mysql_* (obsoleto y eliminado). Usa siempre PDO o MySQLi con sentencias preparadas para prevenir SQL Injection.
conexion.php
<?php
// Configuración de conexión
$host   = 'localhost';
$dbname = 'tienda';
$user   = 'root';
$pass   = '';  // vacía en XAMPP por defecto

try {
    $pdo = new PDO(
        "mysql:host=$host;dbname=$dbname;charset=utf8mb4",
        $user, $pass
    );
    $pdo->setAttribute(PDO::ATTR_ERRMODE, PDO::ERRMODE_EXCEPTION);
    $pdo->setAttribute(PDO::ATTR_DEFAULT_FETCH_MODE, PDO::FETCH_ASSOC);
    echo "✅ Conexión exitosa";
} catch (PDOException $e) {
    die("❌ Error: " . $e->getMessage());
}
?>
▶ Resultado
✅ Conexión exitosa
3
CRUD Completo con PDO
Create, Read, Update, Delete — las 4 operaciones fundamentales.
create.php
<?php
// INSERT con sentencia preparada (previene SQL Injection)
$sql = "INSERT INTO productos (nombre, precio, stock, categoria)
        VALUES (:nombre, :precio, :stock, :categoria)";

$stmt = $pdo->prepare($sql);
$stmt->execute([
    ':nombre'    => 'Monitor 4K',
    ':precio'    => 349.99,
    ':stock'     => 8,
    ':categoria' => 'Electrónica'
]);

$nuevoId = $pdo->lastInsertId();
echo "Producto creado con ID: $nuevoId";
?>
▶ Salida
Producto creado con ID: 3
read.php
<?php
// SELECT — todos los productos
$stmt = $pdo->query("SELECT * FROM productos ORDER BY id DESC");
$productos = $stmt->fetchAll();

foreach ($productos as $p) {
    echo "{$p['nombre']} — S/ {$p['precio']}<br>";
}

// SELECT con filtro — un producto por ID
$stmt = $pdo->prepare("SELECT * FROM productos WHERE id = :id");
$stmt->execute([':id' => 1]);
$producto = $stmt->fetch();
echo $producto['nombre'];
?>
▶ Salida
Monitor 4K — S/ 349.99
Laptop Pro — S/ 1299.99
Teclado Mec. — S/ 89.00
update.php
<?php
$stmt = $pdo->prepare(
    "UPDATE productos SET precio = :precio, stock = :stock WHERE id = :id"
);

$actualizado = $stmt->execute([
    ':precio' => 299.99,
    ':stock'  => 12,
    ':id'     => 3
]);

if ($actualizado) {
    echo "✅ Producto actualizado. Filas afectadas: " . $stmt->rowCount();
}
?>
▶ Salida
✅ Producto actualizado. Filas afectadas: 1
delete.php
<?php
$id = 3;  // ID del producto a eliminar

$stmt = $pdo->prepare("DELETE FROM productos WHERE id = :id");
$stmt->execute([':id' => $id]);

if ($stmt->rowCount() > 0) {
    echo "🗑 Producto ID $id eliminado correctamente";
} else {
    echo "⚠️ No se encontró el producto";
}
?>
▶ Salida
🗑 Producto ID 3 eliminado correctamente

Relaciones entre tablas y JOINs
Relaciona tablas para consultas más potentes.
relaciones.sql
-- Tabla usuarios
CREATE TABLE usuarios (
    id     INT AUTO_INCREMENT PRIMARY KEY,
    nombre VARCHAR(100),
    email  VARCHAR(150) UNIQUE
);

-- Tabla pedidos con clave foránea
CREATE TABLE pedidos (
    id           INT AUTO_INCREMENT PRIMARY KEY,
    usuario_id   INT,
    producto_id  INT,
    cantidad     INT DEFAULT 1,
    fecha        TIMESTAMP DEFAULT CURRENT_TIMESTAMP,
    FOREIGN KEY (usuario_id)  REFERENCES usuarios(id),
    FOREIGN KEY (producto_id) REFERENCES productos(id)
);

-- JOIN: obtener pedidos con info de usuario y producto
SELECT
    u.nombre AS usuario,
    p.nombre AS producto,
    p.precio,
    pe.cantidad,
    (p.precio * pe.cantidad) AS total
FROM pedidos pe
INNER JOIN usuarios  u ON pe.usuario_id  = u.id
INNER JOIN productos p ON pe.producto_id = p.id
ORDER BY pe.fecha DESC;
