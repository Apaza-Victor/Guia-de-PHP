<?php
class Producto {
    // Propiedades con visibilidad
    private static int   $contador = 0;   // compartida entre instancias
    public  readonly int $id;            // solo se asigna una vez (PHP 8.1)
    private float $precio;

    public function __construct(
        public string $nombre,           // constructor promotion
        float         $precio,
        public int    $stock = 0
    ) {
        $this->id     = ++self::$contador;
        $this->setPrecio($precio);       // validar al crear
    }

    // Getter
    public function getPrecio(): float {
        return $this->precio;
    }

    // Setter con validación
    public function setPrecio(float $p): void {
        if ($p < 0) throw new InvalidArgumentException("Precio negativo");
        $this->precio = $p;
    }

    // Método de negocio
    public function precioConIGV(float $tasa = 0.18): string {
        return number_format($this->precio * (1 + $tasa), 2);
    }

    public function hayStock(): bool {
        return $this->stock > 0;
    }

    // Método estático de fábrica
    public static function sinStock(string $nombre, float $precio): static {
        return new static($nombre, $precio, 0);
    }

    public static function totalCreados(): int {
        return self::$contador;
    }
}

$p1 = new Producto("Laptop", 999.00, 5);
$p2 = Producto::sinStock("Teclado", 75.00);

echo "#{$p1->id} {$p1->nombre}: S/{$p1->precioConIGV()}";
echo $p1->hayStock() ? "✅ En stock" : "❌ Sin stock";
echo "Total productos: " . Producto::totalCreados();  // 2
?>

herencia.php
<?php
abstract class Vehiculo {
    public function __construct(
        protected string $marca,
        protected int    $anio,
        protected float  $velocidadMax
    ) {}

    // Método concreto — heredado tal cual
    public function info(): string {
        return "{$this->marca} ({$this->anio}) — hasta {$this->velocidadMax} km/h";
    }

    // Método abstracto — cada subclase DEBE implementarlo
    abstract public function tipoCombustible(): string;

    // final — no se puede sobrescribir en subclases
    final public function matricula(): string {
        return strtoupper(substr($this->marca, 0, 3)) . "-" . $this->anio;
    }
}

class Coche extends Vehiculo {
    public function __construct(
        string $marca, int $anio, float $velMax,
        private int $puertas = 4
    ) {
        parent::__construct($marca, $anio, $velMax);  // llamar al padre
    }

    public function tipoCombustible(): string { return "Gasolina"; }

    // Sobrescribir (override) método del padre
    public function info(): string {
        return parent::info() . " | {$this->puertas} puertas";
    }
}

class CocheElectrico extends Coche {
    public function __construct(
        string $marca, int $anio, float $velMax,
        private int $autonomiaKm
    ) {
        parent::__construct($marca, $anio, $velMax);
    }

    public function tipoCombustible(): string { return "Eléctrico"; }

    public function info(): string {
        return parent::info() . " | Autonomía: {$this->autonomiaKm}km";
    }
}

// Polimorfismo — tratar distintos tipos con la misma interfaz
$flota = [
    new Coche("Toyota", 2023, 180),
    new CocheElectrico("Tesla", 2024, 250, 600),
];

foreach ($flota as $v) {
    echo $v->info() . " [{$v->tipoCombustible()}]<br>";
    echo "Matrícula: " . $v->matricula() . "<br>";

    // instanceof — comprobar tipo en runtime
    if ($v instanceof CocheElectrico) {
        echo "⚡ Vehículo eléctrico<br>";
    }
}
?>

metodos_magicos.php
<?php
class Coleccion {
    private array $items = [];

    // __construct — al instanciar
    public function __construct(array $items = []) {
        $this->items = $items;
    }

    // __destruct — al destruir el objeto o terminar el script
    public function __destruct() {
        echo "Colección liberada<br>";
    }

    // __toString — al convertir a string (echo $obj)
    public function __toString(): string {
        return "Colección(" . count($this->items) . " items)";
    }

    // __get — acceder a propiedad no existente/privada
    public function __get(string $clave): mixed {
        return $this->items[$clave] ?? null;
    }

    // __set — asignar a propiedad no existente/privada
    public function __set(string $clave, mixed $valor): void {
        $this->items[$clave] = $valor;
    }

    // __isset — isset($obj->propiedad)
    public function __isset(string $clave): bool {
        return isset($this->items[$clave]);
    }

    // __unset — unset($obj->propiedad)
    public function __unset(string $clave): void {
        unset($this->items[$clave]);
    }

    // __invoke — llamar al objeto como función
    public function __invoke(string $clave): mixed {
        return $this->items[$clave] ?? null;
    }

    // __clone — personalizar comportamiento al clonar
    public function __clone() {
        $this->items = $this->items;  // copia superficial automática
    }

    // __debugInfo — personalizar var_dump()
    public function __debugInfo(): array {
        return ['total' => count($this->items), 'items' => $this->items];
    }
}

$c = new Coleccion();
$c->nombre = "PHP";       // __set
echo $c->nombre;          // __get  → PHP
echo $c;                  // __toString → Colección(1 items)
echo $c("nombre");        // __invoke → PHP
var_dump(isset($c->nombre)); // __isset → bool(true)
$copia = clone $c;        // __clone
?>

fluent_builder.php
<?php
// Patrón Builder con interfaz fluida (method chaining)
// Cada método devuelve $this para encadenar llamadas
class ConsultaSQL {
    private string  $tabla     = "";
    private array   $campos    = ["*"];
    private array   $condiciones = [];
    private ?int   $limite    = null;
    private ?string $orden    = null;

    public function tabla(string $t): static {
        $this->tabla = $t;
        return $this;       // ← clave del encadenamiento
    }

    public function seleccionar(string ...$campos): static {
        $this->campos = $campos;
        return $this;
    }

    public function donde(string $cond): static {
        $this->condiciones[] = $cond;
        return $this;
    }

    public function ordenarPor(string $col, string $dir = "ASC"): static {
        $this->orden = "$col $dir";
        return $this;
    }

    public function limite(int $n): static {
        $this->limite = $n;
        return $this;
    }

    public function construir(): string {
        $sql = "SELECT " . implode(", ", $this->campos);
        $sql .= " FROM {$this->tabla}";
        if ($this->condiciones) {
            $sql .= " WHERE " . implode(" AND ", $this->condiciones);
        }
        if ($this->orden)  $sql .= " ORDER BY {$this->orden}";
        if ($this->limite) $sql .= " LIMIT {$this->limite}";
        return $sql;
    }
}

// Uso fluido — cada método devuelve el objeto para encadenar
$sql = (new ConsultaSQL())
    ->tabla("usuarios")
    ->seleccionar("id", "nombre", "email")
    ->donde("activo = 1")
    ->donde("edad >= 18")
    ->ordenarPor("nombre")
    ->limite(10)
    ->construir();

echo $sql;
// SELECT id, nombre, email FROM usuarios
// WHERE activo = 1 AND edad >= 18
// ORDER BY nombre ASC LIMIT 10
?>
