aritmeticos.php
<?php
$a = 10;  $b = 3;
echo $a +  $b;   // 13    — Suma
echo $a -  $b;   // 7     — Resta
echo $a *  $b;   // 30    — Multiplicación
echo $a /  $b;   // 3.333 — División
echo $a %  $b;   // 1     — Módulo (resto de la división entera)
echo $a ** $b;   // 1000  — Potencia (10³)

// intdiv — división entera (sin decimales)
echo intdiv(10, 3);   // 3
echo fmod(10.5, 3.2); // 0.9 — módulo para floats
echo abs(-42);         // 42  — valor absoluto
echo round(3.567, 2);  // 3.57 — redondear a 2 decimales
echo ceil(4.1);         // 5   — redondear arriba
echo floor(4.9);        // 4   — redondear abajo
echo sqrt(16);          // 4   — raíz cuadrada
echo pow(2, 8);          // 256 — equivale a 2**8

// Incremento / decremento
$n = 5;
echo $n++;   // 5 — usa $n LUEGO incrementa
echo $n;     // 6
echo ++$n;   // 7 — incrementa LUEGO usa $n
echo $n--;   // 7 — usa $n LUEGO decrementa
echo $n;     // 6

// Concatenar strings con .
echo "Hola" . " " . "Mundo";  // Hola Mundo
$s  = "PHP";
$s .= " 8.3";  // "PHP 8.3" — concatenar y asignar
?>
▶ Salida
13  |  7  |  30  |  3.333  |  1  |  1000
intdiv(10,3)=3  |  round(3.567,2)=3.57  |  sqrt(16)=4
$n++ → 5 luego 6  |  ++$n → 7 inmediatamente
comparacion.php
<?php
$x = 5;  $y = "5";
var_dump($x ==  $y);   // true  — igual, comparación laxa (convierte tipos)
var_dump($x === $y);   // false — idéntico: valor Y tipo deben coincidir
var_dump($x !=  $y);   // false — no igual (laxa)
var_dump($x !== $y);   // true  — no idéntico
var_dump(10 >   5);     // true
var_dump(10 >=  10);   // true
var_dump(3  <   7);     // true
var_dump(3  <=  3);     // true

// Operador Spaceship <=> (PHP 7+)
// Devuelve -1, 0, o 1
echo 1 <=> 2;   // -1 ($a < $b)
echo 2 <=> 2;   //  0 ($a == $b)
echo 3 <=> 2;   //  1 ($a > $b)

// Muy útil en usort
$nombres = ["Carlos", "Ana", "Beatriz"];
usort($nombres, fn($a, $b) => $a <=> $b);
// ["Ana", "Beatriz", "Carlos"]

// TRAMPAS comunes con comparación laxa ==
var_dump(0   == "hola");   // true en PHP 7, false en PHP 8 ← cambio importante
var_dump(""  == null);    // true
var_dump("1" == "01");    // true (ambos son 1 numéricamente)
var_dump(100 == "1e2");  // true (1e2 = 100)
// Regla: usa === siempre que puedas
?>
▶ Trampas de ==
5 == "5" → true (laxa) pero 5 === "5" → false (estricta)
"1" == "01" → true ← sorpresa
Spaceship: 1<=>2 = -1  |  2<=>2 = 0  |  3<=>2 = 1
logicos.php
<?php
$a = true;  $b = false;

// Operadores lógicos (forma símbolo — mayor precedencia)
var_dump($a && $b);   // false — AND: ambos deben ser true
var_dump($a || $b);   // true  — OR: al menos uno true
var_dump(!$a);          // false — NOT: invierte

// Operadores lógicos (forma palabra — menor precedencia)
var_dump($a and $b);  // false
var_dump($a or  $b);  // true
var_dump($a xor $b);  // true — XOR: uno pero no ambos

// Short-circuit: PHP deja de evaluar si el resultado ya está claro
function caro(): bool { echo "[se ejecutó]"; return true; }

false && caro();  // [se ejecutó] NO aparece — el && ya sabe que es false
true  || caro();  // [se ejecutó] NO aparece — el || ya sabe que es true

// Aplicación práctica: ejecutar solo si el anterior fue exitoso
file_exists("config.php") && require "config.php";
connect() || throw new RuntimeException("Sin conexión");

// Diferencia && vs and (precedencia)
$r1 = true  &&  false;  // $r1 = (true && false) = false
$r2 = true  and false;  // $r2 = true; (luego: true and false — ignorado)
var_dump($r1, $r2);      // bool(false), bool(true) ← distintos!
?>
▶ Salida clave
$r1 (&&) = false  |  $r2 (and) = true ← misma expresión, distinto resultado
Short-circuit: false && caro() → caro() nunca se llama
asignacion_bits.php
<?php
// Asignación compuesta — operación + asignación en uno
$n = 10;
$n +=  5;   // $n = $n + 5  = 15
$n -=  3;   // $n = $n - 3  = 12
$n *=  2;   // $n = $n * 2  = 24
$n /=  4;   // $n = $n / 4  = 6
$n %=  4;   // $n = $n % 4  = 2
$n **= 3;   // $n = $n ** 3 = 8
$s  = "Hola";
$s .= " PHP"; // $s = "Hola PHP"

// Asignación múltiple / desestructuración
[$x, $y, $z] = [1, 2, 3];        // $x=1, $y=2, $z=3
["a" => $val] = ["a" => 42];   // $val=42

// Intercambiar variables sin temporal
[$x, $y] = [$y, $x];

// OPERADORES BITWISE — trabajan bit a bit
echo 0b1010 &  0b1100;  // 8  (1000) — AND bit a bit
echo 0b1010 |  0b1100;  // 14 (1110) — OR bit a bit
echo 0b1010 ^  0b1100;  // 6  (0110) — XOR bit a bit
echo ~5;                 // -6       — NOT bit a bit
echo 1  << 3;            // 8  — desplazamiento izq. (1 * 2³)
echo 16 >> 2;            // 4  — desplazamiento der. (16 / 2²)

// Uso práctico de bits: flags/permisos
const LEER    = 0b001;  // 1
const ESCRIBIR = 0b010;  // 2
const EJECUTAR = 0b100;  // 4

$permisos = LEER | ESCRIBIR;  // 3 — tiene leer Y escribir
echo ($permisos & LEER)    ? "✅ Puede leer"    : "❌";
echo ($permisos & EJECUTAR) ? "✅ Puede ejecutar" : "❌ No ejecutar";
?>
▶ Salida
✅ Puede leer
❌ No ejecutar
Desplazamiento: 1<<3 = 8  |  16>>2 = 4
