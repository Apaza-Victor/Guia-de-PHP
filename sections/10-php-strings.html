strings_basic.php
<?php
$s = "  Hola, Mundo PHP!  ";

// Longitud y espacios
echo strlen($s);       // 20 — en bytes (cuidado con UTF-8)
echo mb_strlen($s);   // 20 — en caracteres (multibyte seguro)
echo trim($s);        // "Hola, Mundo PHP!" — quita espacios de ambos lados
echo ltrim($s);       // quita solo por la izquierda
echo rtrim($s);       // quita solo por la derecha

// Mayúsculas / minúsculas
echo strtoupper($s);   // HOLA, MUNDO PHP!
echo strtolower($s);   // hola, mundo php!
echo ucfirst("hola mundo");  // Hola mundo
echo ucwords("hola mundo php"); // Hola Mundo Php
echo mb_strtoupper("ñoño");    // ÑOÑO — seguro con acentos

// Extracción
echo substr($s, 2, 4);    // "Hola" — desde pos 2, longitud 4
echo substr($s, -4);      // "PHP!" — últimos 4 caracteres
echo mb_substr("Ñoño", 0, 2); // "Ño" — seguro con UTF-8

// Dividir y unir
$partes = explode(",", "a,b,c,d");     // ["a","b","c","d"]
$partes = explode(",", "a,b,c,d", 2);  // ["a","b,c,d"] — max 2 partes
echo implode(" | ", ["PHP","HTML","CSS"]); // PHP | HTML | CSS

// Repetir y rellenar
echo str_repeat("=-", 5);             // =-=-=-=-
echo str_pad("42", 6, "0", STR_PAD_LEFT); // 000042
echo wordwrap("Lorem ipsum dolor", 10, "\n");  // corta a 10 chars
?>
▶ Salida destacada
trim(): "Hola, Mundo PHP!"
substr($s, -4): "PHP!"
implode: PHP | HTML | CSS
str_pad("42",6,"0",LEFT): 000042
strings_buscar.php
<?php
$texto = "El veloz murciélago hindú comía feliz cardillo y kiwi";

// Buscar posición
echo strpos($texto,  "kiwi");    // posición primera ocurrencia
echo strrpos($texto, "i");      // posición ÚLTIMA ocurrencia de "i"
echo stripos($texto,  "KIWI");   // sin distinguir mayúsculas

// Comprobar presencia (PHP 8) — más legible que strpos
echo str_contains($texto,   "kiwi");    // true
echo str_starts_with($texto, "El");      // true
echo str_ends_with($texto,   "kiwi");    // true

// Contar ocurrencias
echo substr_count($texto, "i");   // cuántas veces aparece "i"

// Reemplazar
echo str_replace("kiwi", "mango", $texto);
echo str_ireplace("KIWI", "mango", $texto); // insensible a mayúsculas

// Reemplazar múltiples a la vez
$limpio = str_replace(
    ["veloz", "hindú", "feliz"],
    ["rápido", "español", "contento"],
    $texto
);

// Extraer parte después de una subcadena
echo strstr($texto, "kiwi");         // "kiwi" en adelante
echo strstr($texto, "kiwi", true);  // todo ANTES de "kiwi"

// Invertir y mezclar
echo strrev("PHP");  // PHP — PHP es palíndromo!
?>
▶ Salida
str_contains("kiwi") → true
str_starts_with("El") → true
str_ends_with("kiwi") → true
strrev("PHP") → PHP ← ¡palíndromo!
strings_format.php
<?php
// sprintf — formato de cadena (no imprime, devuelve)
$precio = sprintf("S/ %.2f", 49.9);      // "S/ 49.90"
$codigo = sprintf("%05d", 42);          // "00042"
$hex    = sprintf("%x", 255);            // "ff"
$msg    = sprintf("Hola %s, tienes %d años", "Ana", 28);

// printf — igual que sprintf pero sí imprime
printf("%.2f%%", 98.6);  // 98.60% — %% imprime el símbolo %

// Formatear números
echo number_format(1234567.891, 2);          // 1,234,567.89
echo number_format(1234567.891, 2, ",", ".");  // 1.234.567,89 (español)
echo money_format("%.2n", 1234.56);          // depende del locale

// Convertir entre bases numéricas
echo decbin(10);    // "1010"  — decimal a binario
echo decoct(8);     // "10"    — decimal a octal
echo dechex(255);   // "ff"    — decimal a hexadecimal
echo bindec("1010"); // 10     — binario a decimal
echo hexdec("ff");  // 255    — hexadecimal a decimal

// Caracteres ASCII y Unicode
echo chr(65);        // "A"   — ASCII a caracter
echo ord("A");       // 65    — caracter a ASCII
echo mb_chr(0x00F1); // "ñ"   — Unicode a caracter

// Hash de texto
echo md5("hola");           // hash MD5 (no usar para passwords)
echo sha1("hola");          // hash SHA-1
echo hash("sha256", "hola"); // SHA-256
?>
▶ Salida

decbin(10) → 1010  |  dechex(255) → ff

heredoc_nowdoc.php
<?php
$nombre = "Ana";
$total  = 299.90;

// HEREDOC — como comillas dobles: interpola variables
$email = <<<HTML
<html>
  <body>
    <h1>Hola, $nombre</h1>
    <p>Tu pedido de S/ $total está listo.</p>
    <a href="https://tienda.com/pedidos">Ver pedido</a>
  </body>
</html>
HTML;

echo $email;

// NOWDOC — como comillas simples: NO interpola
$plantilla = <<<'TMPL'
Hola, $nombre.
Tu total es: $total.
(Esto es una plantilla sin interpolar)
TMPL;

echo $plantilla;
// Hola, $nombre.
// Tu total es: $total.

// Diferencias clave:
// Heredoc  → <<
// Nowdoc   → <<<'LABEL' → texto literal, sin interpolación

// Heredoc en llamada a función (PHP 7.3+)
echo trim(<<<EOT
  Texto con espacios extra
  en varias líneas
EOT);

// Heredoc dentro de array
$config = [
    "query" => <<<SQL
        SELECT *
        FROM usuarios
        WHERE activo = 1
        ORDER BY nombre
SQL,
];
?>
▶ Heredoc vs Nowdoc
Heredoc: "Hola, Ana." — variable interpolada
Nowdoc: "Hola, $nombre." — texto literal
Heredoc <<<LABELNowdoc <<<'LABEL'
Interpolación✅ Sí ($var, {$obj->p})❌ No — texto literal
Equivale aComillas dobles "..."Comillas simples '...'
