variables.php
<?php
$nombre    = "Ana García";   // string
$edad      = 25;             // int
$altura    = 1.68;           // float
$activo    = true;           // bool
$direccion = null;           // null
$colores   = ["rojo", "azul"]; // array

// Ver tipo de una variable
echo gettype($nombre);    // string
echo gettype($edad);      // integer

// var_dump — tipo + valor + longitud
var_dump($activo);        // bool(true)
var_dump($nombre);        // string(10) "Ana García"

// Interpolación dentro de comillas dobles
echo "Nombre: $nombre, Edad: $edad";
echo "Altura: {$altura}m";  // {} para expresiones complejas

// isset(), empty(), is_null()
var_dump(isset($nombre));    // true — existe y no es null
var_dump(empty($direccion)); // true — null es "vacío"
var_dump(is_string($nombre)); // true
var_dump(is_int($edad));      // true
?>
▶ Salida
string  |  integer
bool(true)  |  string(10) "Ana García"
Nombre: Ana García, Edad: 25
casting.php
<?php
// Casting explícito — forzar un tipo
$precio = "49.99abc";

echo (int)   $precio;     // 49     — toma la parte entera
echo (float) $precio;     // 49.99  — toma la parte decimal
echo (bool)  $precio;     // true   — string no vacío es true
echo (string) 3.14;       // "3.14"
echo (int) true;          // 1
echo (int) false;         // 0
echo (float) "3,14";      // 3.0 — coma no es separador decimal

// Funciones de conversión
echo intval("0xFF", 16);  // 255 — hex a decimal
echo intval("0b1010", 2); // 10  — binario a decimal
echo floatval("3.14xyz"); // 3.14
echo strval(42);          // "42"
echo settype($precio, "float"); // convierte $precio in-place

// Valores "falsy" en PHP — se evalúan como false
$falsys = [0, 0.0, "", "0", null, false, []];
foreach ($falsys as $v) {
    var_dump((bool) $v);  // todos son bool(false)
}

// Cualquier otro valor es "truthy"
var_dump((bool) "false");  // bool(true) — string no vacío!
var_dump((bool) [0]);      // bool(true) — array con un elemento
?>
▶ Salida destacada
("false") → bool(true) ← string no vacío siempre es truthy
([0]) → bool(true) ← array con elementos es truthy
("0") → bool(false) ← excepción: "0" es falsy
constantes.php
<?php
// Constante global con define()
define('VERSION', '1.5.0');
define('IVA', 0.21);
define('COLORES_VALIDOS', ['rojo', 'verde', 'azul']);

echo VERSION;  // 1.5.0  (sin $ y sin paréntesis)
echo IVA;      // 0.21

// Constante de clase con const
class Conexion {
    public const TIMEOUT   = 30;
    public const MAX_RETRY = 3;

    public function conectar(): void {
        echo "Timeout: " . self::TIMEOUT;
    }
}
echo Conexion::TIMEOUT;   // 30 — acceso sin instanciar
echo Conexion::MAX_RETRY; // 3

// Constantes mágicas de PHP — cambian según contexto
echo __FILE__;      // ruta completa del archivo actual
echo __DIR__;       // directorio del archivo actual
echo __LINE__;      // número de línea actual
echo __FUNCTION__;  // nombre de la función actual
echo __CLASS__;     // nombre de la clase actual
echo __NAMESPACE__; // namespace actual
?>
▶ Salida
VERSION: 1.5.0  |  IVA: 0.21
Conexion::TIMEOUT = 30
__FILE__: /var/www/html/index.php
scope.php
<?php
$global = "Soy global";

function sinAcceso(): void {
    // ❌ Las funciones NO ven variables globales por defecto
    echo $global;  // Notice: undefined variable
}

function conGlobal(): void {
    // ✅ Importar con global
    global $global;
    echo $global;  // Soy global
}

// Variable estática — se inicializa una vez y persiste entre llamadas
function contarLlamadas(): int {
    static $veces = 0;  // se inicializa solo la primera vez
    return ++$veces;
}

echo contarLlamadas();  // 1
echo contarLlamadas();  // 2
echo contarLlamadas();  // 3

// Variables variables — el nombre es otra variable
$campo = "precio";
$$campo = 99.99;  // crea $precio = 99.99
echo $precio;      // 99.99
echo $$campo;     // 99.99
?>
▶ Salida
sinAcceso(): Notice — variable no definida
conGlobal(): Soy global
contarLlamadas(): 1 → 2 → 3
$$campo: 99.99
