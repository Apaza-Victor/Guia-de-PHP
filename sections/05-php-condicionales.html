Diagrama: IF / ELSEIF / ELSE
if_elseif.php
<?php
$nota = 75;

// Estructura if / elseif / else
if ($nota >= 90) {
    echo "Sobresaliente";
} elseif ($nota >= 70) {
    echo "Notable";
} elseif ($nota >= 50) {
    echo "Aprobado";
} else {
    echo "Suspenso";
}

// if de una sola línea (sin llaves — solo si hay 1 instrucción)
if ($nota > 0) echo "Nota válida";

// Sintaxis alternativa útil en plantillas HTML
if ($nota >= 50):
    echo "<p>Aprobado</p>";
elseif ($nota >= 30):
    echo "<p>Cerca</p>";
else:
    echo "<p>Suspenso</p>";
endif;

// Condiciones compuestas
$edad = 20;
$tieneDocumento = true;
if ($edad >= 18 && $tieneDocumento) {
    echo "Puede ingresar";
}
?>
▶ Salida
Notable
Nota válida
Puede ingresar
Diagrama de flujo: switch
switch.php
<?php
$dia = "lunes";

// switch usa comparación laxa (==), no estricta (===)
switch ($dia) {
    case "lunes":
    case "martes":
    case "miércoles":
        echo "Primera mitad de semana";
        break;  // sin break, cae al siguiente case
    case "jueves":
    case "viernes":
        echo "Segunda mitad de semana";
        break;
    case "sábado":
    case "domingo":
        echo "¡Fin de semana!";
        break;
    default:
        echo "Día no válido";
}

// switch con return dentro de función — sin break necesario
function tipoUsuario(int $rol): string {
    switch ($rol) {
        case 1:  return "Administrador";
        case 2:  return "Editor";
        case 3:  return "Lector";
        default: return "Desconocido";
    }
}
echo tipoUsuario(2);  // Editor
?>
▶ Salida
Primera mitad de semana
Editor
Cuidado: Sin break, el switch "cae" (fall-through) al siguiente case. Esto puede ser intencional (agrupar casos) o un bug.
match.php
<?php
// match usa comparación ESTRICTA (===), es una expresión (devuelve valor)
$codigo = 404;

$mensaje = match($codigo) {
    200       => "✅ OK",
    201       => "✅ Creado",
    301, 302  => "↩ Redirección",
    400       => "❌ Petición incorrecta",
    401       => "🔒 No autorizado",
    403       => "🚫 Prohibido",
    404       => "🔍 No encontrado",
    500       => "💥 Error del servidor",
    default   => "Código desconocido"
};
echo $mensaje;  // 🔍 No encontrado

// match con condición booleana (no valor)
$nota = 75;
$calificacion = match(true) {
    $nota >= 90 => "Sobresaliente",
    $nota >= 70 => "Notable",
    $nota >= 50 => "Aprobado",
    default     => "Suspenso"
};
echo $calificacion;  // Notable

// match lanza UnhandledMatchError si ningún caso aplica y no hay default
// Por eso siempre añade default cuando sea posible
?>

operadores_condicionales.php
<?php
// Ternario: condición ? valor_true : valor_false
$edad = 20;
$acceso = $edad >= 18 ? "Permitido" : "Denegado";

// Elvis ?: — si el valor ya es truthy, úsalo; si no, usa el alternativo
$nombre = "";
$display = $nombre ?: "Anónimo";   // "Anónimo" (porque "" es falsy)

// Null Coalescing ?? — si existe y no es null, úsalo; si no, usa el alternativo
$usuario = $_GET['user'] ?? "invitado";   // seguro aunque no exista $_GET['user']
$config  = $a ?? $b ?? $c ?? "default"; // encadenado

// ??= Null Coalescing Assignment (PHP 7.4) — asigna solo si es null
$config['timeout'] ??= 30;  // solo asigna 30 si 'timeout' no estaba definido

// Nullsafe operator ?-> (PHP 8) — evita errores al encadenar objetos nulables
class Direccion {
    public function __construct(public string $ciudad) {}
}
class Perfil {
    public ?Direccion $direccion = null;
}
class Usuario {
    public ?Perfil $perfil = null;
}

$u = new Usuario();

// ❌ Sin nullsafe — crash si $perfil es null
// echo $u->perfil->direccion->ciudad;

// ✅ Con nullsafe — devuelve null si cualquier parte es null
$ciudad = $u?->perfil?->direccion?->ciudad ?? "Sin ciudad";
echo $ciudad;  // Sin ciudad
?>
▶ Salida
Permitido
Anónimo
Sin ciudad
