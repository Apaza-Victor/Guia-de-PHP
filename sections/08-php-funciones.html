funciones_tipos.php
<?php
// declare(strict_types=1) — obliga a pasar exactamente el tipo declarado
declare(strict_types=1);

// Función con tipos de entrada y retorno
function sumar(int $a, int $b): int {
    return $a + $b;
}
echo sumar(3, 4);  // 7

// Tipo nullable — puede recibir null además del tipo declarado
function saludar(?string $nombre): string {
    return "Hola, " . ($nombre ?? "invitado") . "!";
}
echo saludar(null);    // Hola, invitado!
echo saludar("Ana");   // Hola, Ana!

// Tipos unión — acepta varios tipos (PHP 8)
function formatear(int|float $num): string {
    return "S/ " . number_format($num, 2);
}
echo formatear(1500);       // S/ 1,500.00
echo formatear(49.99);     // S/ 49.99

// void — función que no retorna valor
function registrar(string $msg): void {
    error_log($msg);
    // return; — sin valor, o simplemente no poner return
}

// never — función que NUNCA retorna (lanza excepción o termina el script)
function error404(): never {
    http_response_code(404);
    exit("Página no encontrada");
}

// mixed — acepta y/o retorna cualquier tipo
function obtenerValor(string $clave): mixed {
    $datos = ['nombre' => 'Ana', 'edad' => 30, 'activo' => true];
    return $datos[$clave] ?? null;
}
?>
▶ Salida
7

S/ 1,500.00  |  S/ 49.99

variadic_referencia.php
<?php
// Variadic — número variable de argumentos con ...
function sumarTodo(int ...$nums): int {
    return array_sum($nums);  // $nums es un array
}
echo sumarTodo(1, 2, 3);         // 6
echo sumarTodo(10, 20, 30, 40);  // 100

// Mezcla de parámetros normales y variadic
function etiqueta(string $tag, string ...$clases): string {
    $cls = implode(" ", $clases);
    return "<$tag class='$cls'>";
}
echo etiqueta("div", "card", "flex", "p-4");
// <div class='card flex p-4'>

// Spread operator — pasar array como argumentos
$valores = [5, 10, 15];
echo sumarTodo(...$valores);  // 30 — equivale a sumarTodo(5,10,15)

// Paso por REFERENCIA con & — modifica la variable original
function duplicar(int &$valor): void {
    $valor *= 2;  // modifica la variable del exterior
}
$precio = 50;
duplicar($precio);
echo $precio;  // 100 — cambió el original

// Sin referencia — solo cambia la copia local
function triplicarCopia(int $valor): int {
    return $valor * 3;
}
$n = 10;
echo triplicarCopia($n);  // 30
echo $n;                   // 10 — $n no cambió
?>
▶ Salida
6  |  100
<div class='card flex p-4'>
$precio tras duplicar(): 100
$n sin referencia sigue siendo: 10
closures_arrows.php
<?php
// Función anónima (closure) — se asigna a variable
$doble = function(int $n): int {
    return $n * 2;
};
echo $doble(5);  // 10

// Capturar variables externas con use
$impuesto = 0.18;
$conIGV = function(float $precio) use ($impuesto): float {
    return $precio * (1 + $impuesto);
};
echo $conIGV(100);  // 118

// Arrow function fn() — captura variables externas automáticamente
$factor = 3;
$multiplicar = fn(int $n) => $n * $factor;  // $factor capturado automáticamente
echo $multiplicar(7);  // 21

// Closures como callbacks — array_map, array_filter, usort
$precios = [100, 250, 50, 400, 30];

$conIGVTodos = array_map(fn($p) => $p * 1.18, $precios);
$caros       = array_filter($precios, fn($p) => $p > 100);
usort($precios, fn($a, $b) => $a <=> $b);  // <=> operador spaceship

print_r($caros);   // [250, 400]
print_r($precios); // [30, 50, 100, 250, 400]

// callable type hint — acepta cualquier función invocable
function aplicar(callable $fn, int $valor): int {
    return $fn($valor);
}
echo aplicar(fn($n) => $n ** 2, 5);  // 25 — fn al cuadrado
echo aplicar('abs', -10);             // 10 — función nativa como callable
?>
▶ Salida
10  |  118  |  21
Caros (>100): [250, 400]
Ordenado: [30, 50, 100, 250, 400]
recursividad.php
<?php
// Factorial — función que se llama a sí misma
function factorial(int $n): int {
    if ($n <= 1) return 1;  // caso base — SIEMPRE necesario
    return $n * factorial($n - 1);  // llamada recursiva
}
echo factorial(5);  // 120  →  5*4*3*2*1
echo factorial(0);  // 1

// Recorrer árbol/menú anidado — muy común en CMS y menús
function renderMenu(array $items, int $nivel = 0): void {
    foreach ($items as $item) {
        echo str_repeat("  ", $nivel) . "- {$item['nombre']}<br>";
        if (!empty($item['hijos'])) {
            renderMenu($item['hijos'], $nivel + 1);
        }
    }
}

$menu = [
    ['nombre' => 'Inicio', 'hijos' => []],
    ['nombre' => 'Productos', 'hijos' => [
        ['nombre' => 'Electrónica', 'hijos' => [
            ['nombre' => 'Laptops', 'hijos' => []],
        ]],
        ['nombre' => 'Ropa', 'hijos' => []],
    ]],
];
renderMenu($menu);

// Fibonacci con memoización — evitar recálculos
function fib(int $n, array &$cache = []): int {
    if ($n <= 1) return $n;
    if (isset($cache[$n])) return $cache[$n];
    return $cache[$n] = fib($n-1, $cache) + fib($n-2, $cache);
}
echo fib(10);  // 55
?>
▶ Salida
factorial(5) = 120
- Inicio
- Productos
  - Electrónica
    - Laptops
  - Ropa
fib(10) = 55
Toda función recursiva necesita un caso base que detenga la recursión. Sin él, causará un error de desbordamiento de pila (stack overflow).
include_require.php
<?php
// include — incluye un archivo. Si falla, solo warning y sigue ejecutando
include 'config.php';

// require — incluye un archivo. Si falla, FATAL ERROR y detiene el script
require 'conexion.php';

// include_once — solo incluye una vez (evita redeclaración de funciones/clases)
include_once 'funciones_helpers.php';

// require_once — lo mismo que require, pero solo una vez
require_once 'vendor/autoload.php';

// Diferencia clave:
//   include  → warning si falla, el script CONTINÚA
//   require  → FATAL ERROR si falla, el script SE DETIENE
//   _once    → evita incluir el mismo archivo dos veces

// Ejemplo práctico — estructura de proyecto
require_once 'config/database.php';  // conexión PDO
require_once 'config/helpers.php';    // funciones auxiliares
include 'templates/header.php';      // cabecera HTML (puede fallar sin gravedad)
// ... contenido principal ...
include 'templates/footer.php';      // pie de página

// Autoload moderno — Composer (recomendado)
require_once 'vendor/autoload.php';
// Ya NO necesitas include/require por cada clase
// Composer carga las clases automáticamente bajo demanda (PSR-4)
?>
Comportamientoincluderequire

En proyectos modernos con Composer, usa require_once 'vendor/autoload.php' una sola vez y olvídate de include/require por cada clase. El autoloader de Composer las carga bajo demanda.
generators.php
<?php
// Generators — producen valores bajo demanda con yield
// Ventaja: memoria O(1) vs O(n) con arrays

// Función normal — carga TODO el array en memoria
function rangoArray(int $start, int $end): array {
    $result = [];
    for ($i = $start; $i <= $end; $i++) {
        $result[] = $i;
    }
    return $result; // Crea el array completo en RAM
}

// Generator — produce valores uno a uno (lazy evaluation)
function rango(int $start, int $end): Generator {
    for ($i = $start; $i <= $end; $i++) {
        yield $i; // produce un valor, pausa, espera a que se pida el siguiente
    }
}

// Se usa igual que un array en foreach
foreach (rango(1, 5) as $num) {
    echo $num . " ";  // 1 2 3 4 5
}

// Generator con yield y return (el return es el valor final del generator)
function leerLineas(string $archivo): Generator {
    $handle = fopen($archivo, 'r');
    if (!$handle) return;
    while (!feof($handle)) {
        yield fgets($handle);
    }
    fclose($handle);
}

// yield clave => valor — como un array asociativo
function usuarios(): Generator {
    yield 'ana' => 'Ana García';
    yield 'luis' => 'Luis Pérez';
    yield 'maria' => 'María López';
}

foreach (usuarios() as $key => $nombre) {
    echo "$key: $nombre\n";
}
?>
▶ Salida
1 2 3 4 5
ana: Ana García
luis: Luis Pérez
maria: María López
Un Generator es como una función que pausa y reanuda su ejecución. Cada vez que el foreach pide un valor, el generator ejecuta hasta el siguiente yield. Ideal para procesar archivos grandes, series de datos o flujos infinitos sin cargar todo en memoria.
