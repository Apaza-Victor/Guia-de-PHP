Interfaces y Contratos
Una interfaz define qué métodos debe implementar una clase, sin definir cómo.
Las interfaces son contratos: garantizan que una clase tendrá ciertos métodos. Una clase puede implementar múltiples interfaces.
interfaces.php
<?php
// Definir interfaz
interface Pagable {
    public function pagar(float $monto): bool;
    public function obtenerSaldo(): float;
}

interface Reembolsable {
    public function reembolsar(float $monto): bool;
}

// Implementar múltiples interfaces
class TarjetaCredito implements Pagable, Reembolsable {
    private float $saldo;

    public function __construct(float $saldoInicial) {
        $this->saldo = $saldoInicial;
    }

    public function pagar(float $monto): bool {
        if ($monto > $this->saldo) return false;
        $this->saldo -= $monto;
        return true;
    }

    public function obtenerSaldo(): float { return $this->saldo; }

    public function reembolsar(float $monto): bool {
        $this->saldo += $monto;
        return true;
    }
}

$tarjeta = new TarjetaCredito(500.00);
$tarjeta->pagar(120.50);
echo "Saldo: S/ " . $tarjeta->obtenerSaldo();  // 379.50

// Type hint con interfaz — acepta cualquier clase que la implemente
function procesarPago(Pagable $metodo, float $monto): void {
    if ($metodo->pagar($monto)) {
        echo "✅ Pago de S/ $monto realizado";
    } else {
        echo "❌ Saldo insuficiente";
    }
}
?>
▶ Salida
Saldo: S/ 379.5
Pago de S/ 50 realizado

Traits — Reutilización de código
Los Traits permiten compartir métodos entre clases sin herencia. Como "mixins".
traits.php
<?php
// Trait reutilizable
trait Timestamps {
    private string $creadoEn;
    private string $actualizadoEn;

    public function setCreado(): void {
        $this->creadoEn = date('Y-m-d H:i:s');
    }
    public function getCreado(): string { return $this->creadoEn; }
}

trait Auditable {
    public function log(string $msg): void {
        error_log("[AUDIT] $msg");
    }
}

// Usar múltiples traits en una clase
class Producto {
    use Timestamps, Auditable;

    public function __construct(public string $nombre) {
        $this->setCreado();
        $this->log("Producto '$nombre' creado");
    }
}

class Usuario {
    use Timestamps;  // También puede usar el mismo trait
    public function __construct(public string $email) {
        $this->setCreado();
    }
}

$prod = new Producto("Laptop");
echo $prod->getCreado();  // 2024-01-15 10:30:00
?>
▶ Salida
2024-01-15 10:30:00
← error_log registra en archivo de log del servidor
trait_conflict.php
<?php
// Resolución de conflictos entre traits
trait Cacheable {
    public function getTTL(): int { return 3600; }  // 1 hora
    public function 储(): void { echo "Guardando en caché..."; }
}

trait RateLimited {
    public function getTTL(): int { return 60; }     // 1 minuto
    public function getMaxRequests(): int { return 100; }
}

// Cuando dos traits tienen el mismo método → conflicto
// Usar "insteadof" para elegir cuál usar
class MiAPI {
    use Cacheable, RateLimited {
        // Usar getTTL de Cacheable, ignorando el de RateLimited
        Cacheable::getTTL insteadof RateLimited;
        // Renombrar el método del otro trait con "as"
        RateLimited::getTTL as getRateTTL;
    }
}

$api = new MiAPI();
echo $api->getTTL();      // 3600 (de Cacheable)
echo $api->getRateTTL();  // 60 (de RateLimited, renombrado)
?>

PHP 8 — Novedades en POO
Constructor promotion, readonly, enums y otras características modernas.
constructor_promotion.php
<?php
// ❌ Antes de PHP 8 — repetitivo
class UsuarioViejo {
    public string $nombre;
    public string $email;
    public int $edad;

    public function __construct(string $nombre, string $email, int $edad) {
        $this->nombre = $nombre;
        $this->email  = $email;
        $this->edad   = $edad;
    }
}

// ✅ PHP 8 — Constructor Property Promotion
class Usuario {
    public function __construct(
        public string $nombre,
        public string $email,
        public int    $edad = 18   // valor por defecto
    ) {}  // ¡No hace falta nada más!
}

$u = new Usuario("Ana", "ana@email.com");
echo $u->nombre;  // Ana
echo $u->edad;    // 18
?>
▶ Salida
Ana  |  18
readonly.php
<?php
// readonly — solo se asigna una vez (PHP 8.1)
class Pedido {
    public function __construct(
        public readonly int    $id,
        public readonly string $total,
        public readonly string $fecha
    ) {}
}

$pedido = new Pedido(1, "S/299.90", date('Y-m-d'));
echo $pedido->id;  // 1

// ❌ Esto lanzará un error — los readonly no se pueden modificar
// $pedido->id = 2;  // Error: Cannot modify readonly property

// readonly class (PHP 8.2) — todas las propiedades son readonly
readonly class Punto {
    public function __construct(
        public float $x,
        public float $y
    ) {}
}
?>
▶ Salida
1
enums.php
<?php
// Enum básico (PHP 8.1)
enum Estado {
    case Pendiente;
    case Aprobado;
    case Rechazado;
}

// Backed Enum — con valor asociado
enum Rol: string {
    case Admin   = 'admin';
    case Editor  = 'editor';
    case Lector  = 'lector';

    public function etiqueta(): string {
        return match($this) {
            Rol::Admin  => '👑 Administrador',
            Rol::Editor => '✏️ Editor',
            Rol::Lector => '👁 Lector',
        };
    }
}

$rol = Rol::Admin;
echo $rol->value;      // admin
echo $rol->etiqueta(); // 👑 Administrador

// Crear desde valor
$rol2 = Rol::from('editor');
echo $rol2->etiqueta(); // ✏️ Editor

// Uso como type hint
function asignarRol(Rol $rol): string {
    return "Rol asignado: " . $rol->etiqueta();
}
?>

admin
Administrador

estaticos.php
<?php
class Contador {
    // Propiedad estática — compartida por TODAS las instancias
    private static int $total = 0;

    public function __construct() {
        self::$total++;  // self:: para acceder a miembros de la clase
    }

    // Método estático — se llama sin instanciar la clase
    public static function getTotal(): int {
        return self::$total;
    }

    // Patrón Singleton con estático
    private static ?Contador $instancia = null;
    public static function getInstance(): static {
        if (self::$instancia === null) {
            self::$instancia = new static();
        }
        return self::$instancia;
    }
}

new Contador();
new Contador();
echo Contador::getTotal();  // 2 — sin instanciar
?>
▶ Salida
2

Clases Abstractas
Entre interfaz y clase concreta: define estructura obligatoria con implementación parcial.
Usa clases abstractas cuando quieras compartir código base y obligar a implementar ciertos métodos. Usa interfaces cuando solo quieras definir un contrato.
abstractas.php
<?php
abstract class Notificacion {
    // Método concreto — ya implementado
    public function formatearMensaje(string $msg): string {
        return "[" . date('H:i') . "] " . $msg;
    }

    // Método abstracto — cada subclase DEBE implementarlo
    abstract public function enviar(string $destino, string $mensaje): bool;
}

class NotificacionEmail extends Notificacion {
    public function enviar(string $destino, string $mensaje): bool {
        $msg = $this->formatearMensaje($mensaje);
        echo "📧 Email a $destino: $msg";
        return true;
    }
}

class NotificacionSMS extends Notificacion {
    public function enviar(string $destino, string $mensaje): bool {
        $msg = $this->formatearMensaje($mensaje);
        echo "📱 SMS a $destino: $msg";
        return true;
    }
}

$notif = new NotificacionEmail();
$notif->enviar("user@ejemplo.com", "Tu pedido fue enviado");
?>
▶ Salida
Email a user@ejemplo.com: [10:30] Tu pedido fue enviado
